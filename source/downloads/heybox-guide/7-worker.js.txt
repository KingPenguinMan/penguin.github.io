const telegram = async (env, method, body) => {
  const response = await fetch(`https://api.telegram.org/bot${env.TG_BOT_TOKEN}/${method}`, {
    method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(body),
  });
  const result = await response.json();
  if (!response.ok || !result.ok) throw new Error('Telegram request failed');
  return result;
};
const menu = { inline_keyboard: [
  [{ text: '账号状态', callback_data: 'a1:status' }, { text: '手动签到', callback_data: 'a1:sign' }],
  [{ text: '执行每日任务', callback_data: 'a1:run' }, { text: '抽奖活动', callback_data: 'a1:activities' }],
] };
export default {
  async fetch(request, env) {
    if (request.method === 'GET') return new Response('Heybox Telegram bot is running.');
    if (request.method !== 'POST' || new URL(request.url).pathname !== '/telegram') return new Response('Not found', { status: 404 });
    if (request.headers.get('x-telegram-bot-api-secret-token') !== env.TG_WEBHOOK_SECRET) return new Response('Forbidden', { status: 403 });
    const update = await request.json();
    const callback = update.callback_query;
    const chat = update.message?.chat ?? callback?.message?.chat;
    if (!chat || String(chat.id) !== String(env.TG_CHAT_ID) || chat.type !== 'private') return new Response('OK');
    const id = String(update.update_id ?? '');
    if (!/^\d+$/.test(id) || await env.BOT_STATE.get('update:' + id)) return new Response('OK');
    await env.BOT_STATE.put('update:' + id, '1', { expirationTtl: 86400 });
    const send = text => telegram(env, 'sendMessage', { chat_id: chat.id, text, reply_markup: menu });
    try {
      const text = String(update.message?.text ?? '').trim().split('@')[0];
      const data = callback ? String(callback.data ?? '').replace(/^a1:/, '') : text.slice(1);
      if (!callback && ['start', 'menu', 'help'].includes(data)) await send('小黑盒每日签到\n请选择操作：');
      else {
        let action = ['status', 'sign', 'run', 'activities'].includes(data) ? data : '';
        let activity_id = '', confirmation = '';
        const detail = /^activity:(\d{1,20})$/.exec(data);
        const join = /^join:(\d{1,20}):([A-Za-z0-9_-]{16})$/.exec(data);
        if (callback && detail) { action = 'activity'; activity_id = detail[1]; }
        if (callback && join) { action = 'activity_join'; activity_id = join[1]; confirmation = join[2]; }
        if (!action) await send('请使用菜单选择操作。');
        else {
          const response = await fetch(`https://api.github.com/repos/${env.GITHUB_OWNER}/${env.GITHUB_REPO}/actions/workflows/telegram-command.yml/dispatches`, {
            method: 'POST', headers: { authorization: `Bearer ${env.BOT_GITHUB_TOKEN}`, accept: 'application/vnd.github+json',
              'x-github-api-version': '2022-11-28', 'user-agent': 'heybox-telegram-bot' },
            body: JSON.stringify({ ref: 'main', inputs: { action, activity_id, confirmation } }),
          });
          if (!response.ok) throw new Error('GitHub dispatch failed');
          if (callback) await telegram(env, 'answerCallbackQuery', { callback_query_id: callback.id, text: '已提交，请稍候。' });
          else await send('任务已提交，请稍候。');
        }
      }
    } catch { await send('操作提交失败，请检查部署与令牌配置。'); }
    return new Response('OK');
  },
};
