const TELEGRAM = 'https://api.telegram.org';

function accountName(slot) { return `账号 ${slot}`; }
function menu(slot) {
  return { inline_keyboard: [
    [{ text: `👤 ${accountName(slot)}`, callback_data: 'accounts' }, { text: '📊 账号状态', callback_data: 'status' }],
    [{ text: '✅ 手动签到', callback_data: 'sign' }, { text: '🎯 执行每日任务', callback_data: 'run' }],
    [{ text: '🎁 抽奖活动', callback_data: 'activities' }],
  ].map(row => row.map(button => button.callback_data === 'accounts' ? button : { ...button, callback_data: `a${slot}:${button.callback_data}` })) };
}
function accountMenu(active) {
  return { inline_keyboard: [[1, 2, 3, 4, 5].map(slot => ({ text: `${slot === active ? '✅ ' : ''}${accountName(slot)}`, callback_data: `account:${slot}` }))] };
}
function validAccount(value) { return /^[1-5]$/.test(String(value)); }

async function telegram(env, method, body) {
  const response = await fetch(`${TELEGRAM}/bot${env.TG_BOT_TOKEN}/${method}`, {
    method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(body),
  });
  if (!response.ok) throw new Error(`Telegram ${method} failed`);
  return response.json();
}

async function dispatch(env, action, activityId = '', confirmation = '', account = '1') {
  const response = await fetch(`https://api.github.com/repos/${env.GITHUB_OWNER}/${env.GITHUB_REPO}/actions/workflows/telegram-command.yml/dispatches`, {
    method: 'POST',
    headers: { authorization: `Bearer ${env.BOT_GITHUB_TOKEN}`, accept: 'application/vnd.github+json',
      'x-github-api-version': '2022-11-28', 'user-agent': 'heybox-telegram-bot' },
    body: JSON.stringify({ ref: 'main', inputs: { action, activity_id: activityId, confirmation, account } }),
  });
  if (!response.ok) throw new Error(`GitHub dispatch failed: ${response.status}`);
}

function commandOf(data) {
  if (['status', 'sign', 'run', 'activities'].includes(data)) return { action: data };
  let match = /^activity:(\d{1,20})$/.exec(data);
  if (match) return { action: 'activity', activityId: match[1] };
  match = /^join:(\d{1,20}):([A-Za-z0-9_-]{16})$/.exec(data);
  if (match) return { action: 'activity_join', activityId: match[1], confirmation: match[2] };
}

export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    if (request.method === 'GET') return new Response('Heybox Telegram bot is running.');
    if (request.method !== 'POST' || url.pathname !== '/telegram') return new Response('Not found', { status: 404 });
    if (request.headers.get('x-telegram-bot-api-secret-token') !== env.TG_WEBHOOK_SECRET) return new Response('Forbidden', { status: 403 });
    let update;
    try { update = await request.json(); } catch { return new Response('Bad request', { status: 400 }); }
    const updateId = String(update.update_id ?? '');
    if (!/^\d+$/.test(updateId)) return new Response('OK');
    if (await env.BOT_STATE.get(`update:${updateId}`)) return new Response('OK');
    await env.BOT_STATE.put(`update:${updateId}`, '1', { expirationTtl: 86400 });

    const message = update.message;
    const callback = update.callback_query;
    const chat = message?.chat ?? callback?.message?.chat;
    if (!chat || String(chat.id) !== String(env.TG_CHAT_ID) || chat.type !== 'private') return new Response('OK');
    try {
      let active = await env.BOT_STATE.get(`active-account:${chat.id}`);
      if (!validAccount(active)) active = '1';
      if (message) {
        const text = String(message.text ?? '').split('@')[0];
        if (text === '/start' || text === '/menu' || text === '/help') {
          await telegram(env, 'sendMessage', { chat_id: chat.id, text: `小黑盒任务助手\n当前：${accountName(active)}\n请选择要执行的操作：`, reply_markup: menu(active) });
        } else if (text === '/accounts') {
          await telegram(env, 'sendMessage', { chat_id: chat.id, text: `请选择要管理的账号。未配置的账号会提示配置缺失。`, reply_markup: accountMenu(Number(active)) });
        } else if (['/status', '/sign', '/run', '/activities'].includes(text)) {
          const action = text.slice(1);
          await dispatch(env, action, '', '', active);
          const labels = { status: '正在查询账号状态…', sign: '正在提交手动签到…', run: '正在提交每日任务…', activities: '正在获取抽奖活动…' };
          await telegram(env, 'sendMessage', { chat_id: chat.id, text: `${accountName(active)}：${labels[action]}` });
        } else {
          await telegram(env, 'sendMessage', { chat_id: chat.id, text: '请使用下方菜单操作。', reply_markup: menu(active) });
        }
      } else if (callback) {
        const data = String(callback.data ?? '');
        const account = /^account:([1-5])$/.exec(data);
        if (data === 'accounts') {
          await telegram(env, 'answerCallbackQuery', { callback_query_id: callback.id });
          await telegram(env, 'sendMessage', { chat_id: chat.id, text: '请选择要管理的账号。', reply_markup: accountMenu(Number(active)) });
        } else if (account) {
          active = account[1];
          await env.BOT_STATE.put(`active-account:${chat.id}`, active);
          await telegram(env, 'answerCallbackQuery', { callback_query_id: callback.id, text: `已切换到${accountName(active)}` });
          await telegram(env, 'sendMessage', { chat_id: chat.id, text: `当前已切换到${accountName(active)}。`, reply_markup: menu(active) });
        }
        else {
          const bound = /^a([1-5]):(.+)$/.exec(data);
          const command = bound ? commandOf(bound[2]) : undefined;
          if (!command) await telegram(env, 'answerCallbackQuery', { callback_query_id: callback.id, text: '这个按钮已失效，请重新打开菜单。', show_alert: true });
          else {
            await dispatch(env, command.action, command.activityId, command.confirmation, bound[1]);
            await telegram(env, 'answerCallbackQuery', { callback_query_id: callback.id, text: `${accountName(bound[1])}的任务已提交，请稍候。` });
          }
        }
      }
    } catch (error) {
      if (callback) await telegram(env, 'answerCallbackQuery', { callback_query_id: callback.id, text: '提交失败，请稍后重试。', show_alert: true }).catch(() => {});
      else await telegram(env, 'sendMessage', { chat_id: chat.id, text: '操作提交失败，请稍后重试。' }).catch(() => {});
      console.error(error instanceof Error ? error.message : 'request failed');
    }
    return new Response('OK');
  },
};
